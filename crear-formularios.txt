function crearFormularios() {
  DEFS.forEach(function (d) {
    var form = FormApp.create(d.title);
    form.setDescription(d.purpose + "\n\nLas respuestas son completamente anónimas.");
    form.setCollectEmail(false);
    form.setProgressBar(true);
    d.sections.forEach(function (s, si) {
      form.addPageBreakItem().setTitle(s.title);
      s.qs.forEach(function (q) {
        if (q[1]) form.addMultipleChoiceItem().setTitle(q[0]).setChoiceValues(q[1]).setRequired(true);
        else form.addParagraphTextItem().setTitle(q[0]);
      });
      if (si === d.sections.length - 1) form.addTextItem().setTitle("Nombre (opcional)");
    });
    var hoja = SpreadsheetApp.create(d.title + " (respuestas)");
    form.setDestination(FormApp.DestinationType.SPREADSHEET, hoja.getId());
    if (form.setPublished) form.setPublished(true);
    Logger.log(d.id + " | " + form.getPublishedUrl());
  });
}
