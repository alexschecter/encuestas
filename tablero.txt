function doGet() {
  const props = PropertiesService.getUserProperties();
  let remitentes = [Session.getEffectiveUser().getEmail()];
  try { remitentes = remitentes.concat(GmailApp.getAliases()); } catch (e) {}
  const config = { catalogo: CATALOGO, asunto: props.getProperty("asunto") || ASUNTO_INICIAL, mensaje: props.getProperty("mensaje") || MENSAJE_INICIAL,
    remitentes: remitentes, desde: props.getProperty("desde") || "" };
  return HtmlService.createHtmlOutput(PAGINA.replace("__CONFIG__", JSON.stringify(config).replace(/</g, "\\u003c")))
    .setTitle("Tablero de encuestas")
    .addMetaTag("viewport", "width=device-width, initial-scale=1");
}

function buscarFormulario_(enc) {
  const props = PropertiesService.getScriptProperties();
  const guardado = props.getProperty(enc.id);
  if (guardado) { try { return FormApp.openById(guardado); } catch (e) {} }
  const archivos = DriveApp.getFilesByType(MimeType.GOOGLE_FORMS);
  while (archivos.hasNext()) {
    const f = archivos.next();
    if (f.isTrashed()) continue;
    try {
      const form = FormApp.openById(f.getId());
      if (form.getPublishedUrl().indexOf(enc.publicada) >= 0) { props.setProperty(enc.id, f.getId()); return form; }
    } catch (e) {}
  }
  return null;
}

// Envía el mensaje desde tu cuenta de Gmail, un email por persona.
// "desde" puede ser tu Gmail o cualquier dirección que tengas configurada en Gmail como "Enviar como" (por ejemplo, la de tu dominio).
function enviarEmails(destinatarios, asunto, cuerpo, desde) {
  const lista = String(destinatarios || "").split(/[\s,;]+/).map(function (x) { return x.trim(); }).filter(Boolean);
  const validos = lista.filter(function (x) { return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(x); });
  const invalidos = lista.filter(function (x) { return validos.indexOf(x) < 0; });
  if (!validos.length) throw new Error("No encontré ninguna dirección de email válida.");
  const cupo = MailApp.getRemainingDailyQuota();
  if (validos.length > cupo) throw new Error("Hoy podés enviar " + cupo + " emails más desde el tablero (límite diario de Google).");
  const opciones = {};
  if (desde && GmailApp.getAliases().indexOf(desde) >= 0) opciones.from = desde;
  PropertiesService.getUserProperties().setProperty("desde", desde || "");
  validos.forEach(function (to) { GmailApp.sendEmail(to, asunto, cuerpo, opciones); });
  return JSON.stringify({ enviados: validos.length, invalidos: invalidos });
}

const CAMPO_CLIENTE = "Cliente";

// Ejecutar UNA vez: agrega el campo oculto "Cliente" a los 4 formularios (la encuesta lo completa sola según el enlace).
function agregarCampoCliente() {
  ENCUESTAS.forEach(function (enc) {
    const form = buscarFormulario_(enc);
    if (!form) { Logger.log(enc.id + " | no encontrado"); return; }
    const ya = form.getItems().filter(function (i) { return i.getTitle() === CAMPO_CLIENTE; })[0];
    const item = ya || form.addTextItem().setTitle(CAMPO_CLIENTE).setHelpText("Lo completa la encuesta automáticamente.");
    Logger.log(enc.id + " | listo");
  });
}

function obtenerDatos() {
  return JSON.stringify(ENCUESTAS.map(function (enc) {
    const form = buscarFormulario_(enc);
    if (!form) return { id: enc.id, grupo: enc.grupo, tipo: enc.tipo, titulo: enc.titulo, error: "No encontré este formulario en tu Drive." };
    const respuestas = form.getResponses();
    let seccion = "", idCliente = null;
    const preguntas = [];
    form.getItems().forEach(function (item) {
      const t = item.getType();
      if (t === FormApp.ItemType.PAGE_BREAK) { seccion = item.getTitle(); return; }
      if (item.getTitle() === CAMPO_CLIENTE) { idCliente = item.getId(); return; }
      if (item.getTitle() === "Nombre (opcional)") return; // los nombres no se muestran en el tablero
      if (t === FormApp.ItemType.MULTIPLE_CHOICE) {
        const opciones = item.asMultipleChoiceItem().getChoices().map(function (c) { return c.getValue(); });
        preguntas.push({ id: item.getId(), seccion: seccion, titulo: item.getTitle(), opciones: opciones });
      } else if (t === FormApp.ItemType.PARAGRAPH_TEXT || t === FormApp.ItemType.TEXT) {
        preguntas.push({ id: item.getId(), seccion: seccion, titulo: item.getTitle(), opciones: null });
      }
    });
    const pos = {};
    preguntas.forEach(function (p, i) { pos[p.id] = i; });
    const clientes = {};
    respuestas.forEach(function (r) {
      let cliente = "";
      const its = r.getItemResponses();
      its.forEach(function (ir) { if (ir.getItem().getId() === idCliente) cliente = String(ir.getResponse() || "").trim(); });
      cliente = cliente || "(sin cliente)";
      const c = clientes[cliente] || (clientes[cliente] = { n: 0, ultima: null, conteos: preguntas.map(function (p) { return p.opciones ? p.opciones.map(function () { return 0; }) : null; }), textos: preguntas.map(function (p) { return p.opciones ? null : []; }) });
      c.n++;
      const ts = r.getTimestamp(); if (!c.ultima || ts > c.ultima) c.ultima = ts;
      its.forEach(function (ir) {
        const i = pos[ir.getItem().getId()]; if (i === undefined) return;
        const p = preguntas[i], v = ir.getResponse();
        if (p.opciones) { const j = p.opciones.indexOf(v); if (j >= 0) c.conteos[i][j]++; }
        else if (v && String(v).trim()) c.textos[i].push(String(v).trim());
      });
    });
    Object.keys(clientes).forEach(function (k) { clientes[k].ultima = clientes[k].ultima ? clientes[k].ultima.toISOString() : null; });
    return { id: enc.id, grupo: enc.grupo, tipo: enc.tipo, titulo: enc.titulo, preguntas: preguntas, clientes: clientes };
  }));
}

function generarInforme(cliente, tipo) {
  if (!cliente || cliente === "(sin cliente)") throw new Error("Elegí un cliente en el menú.");
  const raw = JSON.parse(obtenerDatos());
  const d = prepararDatos_(raw, cliente, tipo || "colab");
  const nombreBase = d.clienteNombre + " · " + d.tituloEncuesta + " · " + d.fecha;
  const carpeta = carpetaInformes_();

  const pres = SlidesApp.create(nombreBase + " · Completo");
  const ctx = { pres: pres, d: d, pag: 0, imgs: imagenes_() };
  const inicial = pres.getSlides()[0];
  construirResumen_(ctx);
  const nResumen = ctx.pag;
  construirDetalle_(ctx);
  inicial.remove();
  pres.saveAndClose();
  const fCompleto = DriveApp.getFileById(pres.getId());
  fCompleto.moveTo(carpeta);

  const fResumen = fCompleto.makeCopy(nombreBase + " · Resumen", carpeta);
  const pr = SlidesApp.openById(fResumen.getId());
  pr.getSlides().slice(nResumen).forEach(function (s) { s.remove(); });
  pr.saveAndClose();

  const pdf = function (f) { return carpeta.createFile(f.getAs(MimeType.PDF)).setName(f.getName() + ".pdf").getUrl(); };
  return JSON.stringify({
    resumen: { url: fResumen.getUrl(), pdf: pdf(fResumen) },
    completo: { url: fCompleto.getUrl(), pdf: pdf(fCompleto) },
  });
}

function carpetaInformes_() {

}

// ── Datos ──
function puntos_(j, n) { return n > 1 ? (n - 1 - j) / (n - 1) * 100 : 100; }
function puntaje_(conteos) {
  const n = conteos.reduce(function (a, b) { return a + b; }, 0);
  if (!n) return null;
  return Math.round(conteos.reduce(function (a, c, j) { return a + c * puntos_(j, conteos.length); }, 0) / n);
}
function promedio_(xs) { xs = xs.filter(function (x) { return x !== null && x !== undefined; }); return xs.length ? Math.round(xs.reduce(function (a, b) { return a + b; }, 0) / xs.length) : null; }
function banda_(s) { return s === null ? { t: "Sin datos", c: COL.line } : s >= 75 ? { t: "Fortaleza", c: COL.blue } : s >= 50 ? { t: "En desarrollo", c: COL.sand } : { t: "Prioridad", c: COL.coral }; }

function prepararDatos_(raw, cliente, tipo) {
  const sets = raw.filter(function (s) { return s.tipo === tipo && !s.error; });
  const g = {};
  sets.forEach(function (s) {
    const c = s.clientes[cliente];
    g[s.grupo] = { n: c ? c.n : 0, preguntas: s.preguntas.map(function (p, i) {
      return { seccion: p.seccion, titulo: p.titulo, opciones: p.opciones, conteos: c ? c.conteos[i] : (p.opciones ? p.opciones.map(function () { return 0; }) : null), textos: c ? c.textos[i] : [] };
    }) };
  });
  const grupos = GRUPOS.filter(function (x) { return g[x.k] && g[x.k].n > 0; });
  if (!grupos.length) throw new Error("Todavía no hay respuestas de este cliente en esta encuesta.");
  const base = g[grupos[0].k].preguntas;
  const preguntas = base.map(function (p, i) {
    const q = { seccion: p.seccion, titulo: p.titulo, opciones: p.opciones, por: {} };
    grupos.forEach(function (x) { const pq = g[x.k].preguntas[i]; q.por[x.k] = { conteos: pq.conteos, textos: pq.textos || [], puntaje: p.opciones ? puntaje_(pq.conteos) : null }; });
    if (p.opciones) q.puntaje = promedio_(grupos.map(function (x) { return q.por[x.k].puntaje; }));
    return q;
  });
  const secciones = [];
  preguntas.forEach(function (q) {
    if (!q.opciones) return;
    let s = secciones.filter(function (x) { return x.titulo === q.seccion; })[0];
    if (!s) { s = { titulo: q.seccion, qs: [] }; secciones.push(s); }
    s.qs.push(q);
  });
  secciones.forEach(function (s) {
    s.por = {};
    grupos.forEach(function (x) { s.por[x.k] = promedio_(s.qs.map(function (q) { return q.por[x.k].puntaje; })); });
    s.puntaje = promedio_(grupos.map(function (x) { return s.por[x.k]; }));
  });
  const meses = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
  const hoy = new Date();
  const nombre = String(cliente).replace(/[-_]+/g, " ").replace(/\b\w/g, function (c) { return c.toUpperCase(); });
  return {
    cliente: cliente, clienteNombre: nombre, tipo: tipo, grupos: grupos, n: g, preguntas: preguntas, secciones: secciones,
    tituloEncuesta: tipo === "cultura" ? "Colaboración Interfuncional y Cultura" : "Colaboración Interfuncional",
    fecha: meses[hoy.getMonth()] + " " + hoy.getFullYear(),
  };
}

// ── Dibujo ──
function caja_(sl, text, l, t, w, h, o) {
  o = o || {};
  const s = sl.insertTextBox(text || " ", l, t, w, h);
  const tr = s.getText();
  tr.getTextStyle().setFontFamily(FONT).setFontSize(o.size || 14).setForegroundColor(o.color || COL.ink).setBold(!!o.bold).setItalic(!!o.italic);
  if (o.align) tr.getParagraphStyle().setParagraphAlignment(o.align === "center" ? SlidesApp.ParagraphAlignment.CENTER : o.align === "end" ? SlidesApp.ParagraphAlignment.END : SlidesApp.ParagraphAlignment.START);
  s.setContentAlignment(o.v === "middle" ? SlidesApp.ContentAlignment.MIDDLE : o.v === "bottom" ? SlidesApp.ContentAlignment.BOTTOM : SlidesApp.ContentAlignment.TOP);
  return s;
}
function forma_(sl, tipo, l, t, w, h, color) {
  const s = sl.insertShape(tipo, l, t, w, h);
  s.getFill().setSolidFill(color);
  s.getBorder().setTransparent();
  return s;
}
function rect_(sl, l, t, w, h, color) { return forma_(sl, SlidesApp.ShapeType.RECTANGLE, l, t, w, h, color); }
function tarjeta_(sl, l, t, w, h) { return forma_(sl, SlidesApp.ShapeType.ROUND_RECTANGLE, l, t, w, h, COL.mist); }
function circulo_(sl, l, t, d, color, text, size) {
  const s = forma_(sl, SlidesApp.ShapeType.ELLIPSE, l, t, d, d, color);
  if (text) {
    const tr = s.getText(); tr.setText(text);
    tr.getTextStyle().setFontFamily(FONT).setFontSize(size || 14).setBold(true).setForegroundColor(COL.white);
    tr.getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER);
    s.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE);
  }
  return s;
}
function nueva_(ctx, oscura) {
  const sl = ctx.pres.appendSlide(SlidesApp.PredefinedLayout.BLANK);
  sl.getBackground().setSolidFill(oscura ? COL.navy : COL.white);
  ctx.pag++;
  return sl;
}
function contenido_(ctx, titulo, sub) {
  const sl = nueva_(ctx, false);
  caja_(sl, titulo, 30, 18, 660, 40, { size: 24, bold: true, color: COL.navy });

  caja_(sl, "Encuesta de " + ctx.d.tituloEncuesta + " · " + ctx.d.clienteNombre, 30, 376, 520, 20, { size: 9, color: COL.mute });
  caja_(sl, String(ctx.pag), 640, 376, 50, 20, { size: 9, color: COL.mute, align: "end" });

}
function pct_(c, n) { return n ? Math.round(c / n * 100) : 0; }
function nombreGrupo_(k) { return GRUPOS.filter(function (x) { return x.k === k; })[0].nombre; }
function masElegida_(q, k) {
  const c = q.por[k].conteos; let j = 0;
  c.forEach(function (v, i) { if (v > c[j]) j = i; });
  return q.opciones[j];
}

// Barras horizontales por grupo (una fila por etiqueta)
function barras_(sl, d, filas, l, t, w, h, anchoEtiqueta) {
  const gs = d.grupos;
  const ley = caja_(sl, gs.map(function (x) { return "■ " + x.nombre; }).join("     "), l + anchoEtiqueta, t - 4, w - anchoEtiqueta, 18, { size: 10, color: COL.ink }).getText();
  let pos = 0;
  gs.forEach(function (x) { ley.getRange(pos, pos + 1).getTextStyle().setForegroundColor(x.color); pos += x.nombre.length + 2 + 5; });
  const top = t + 18, alto = (h - 18) / filas.length, barH = Math.min(12, (alto - 10) / gs.length), maxW = w - anchoEtiqueta - 40;
  filas.forEach(function (f, i) {
    const y = top + i * alto;
    caja_(sl, f.etiqueta, l, y, anchoEtiqueta - 8, alto, { size: 10.5, color: COL.ink, align: "end", v: "middle" });
    const bloque = barH * gs.length + 2 * (gs.length - 1), y0 = y + (alto - bloque) / 2;
    gs.forEach(function (x, gi) {
      const v = f.valores[x.k];
      const yy = y0 + gi * (barH + 2);
      if (v !== null && v > 0) rect_(sl, l + anchoEtiqueta, yy, Math.max(2, maxW * v / 100), barH, x.color);
      caja_(sl, v === null ? "–" : String(v), l + anchoEtiqueta + Math.max(2, maxW * (v || 0) / 100) + 2, yy - 4, 36, barH + 8, { size: 9.5, color: COL.ink, v: "middle" });
    });
  });
}

// Barra apilada con el % de cada respuesta
function apilada_(sl, q, k, l, t, w, h) {
  const c = q.por[k].conteos, n = c.reduce(function (a, b) { return a + b; }, 0), cols = ESCALA[c.length] || ESCALA[4];
  if (!n) { rect_(sl, l, t, w, h, COL.line); return; }
  let x = l;
  c.forEach(function (v, j) {
    if (!v) return;
    const ww = w * v / n, s = rect_(sl, x, t, ww, h, cols[j]);
    if (ww >= 26) {
      const tr = s.getText(); tr.setText(pct_(v, n) + "%");
      tr.getTextStyle().setFontFamily(FONT).setFontSize(10).setBold(true).setForegroundColor(j === 0 || j === c.length - 1 ? COL.white : COL.ink);
      tr.getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER);
      s.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE);
    }
    x += ww;
  });
}

// ── Resumen ──
function construirResumen_(ctx) {
  const d = ctx.d, gs = d.grupos, dos = gs.length === 2;
  const secs = d.secciones.slice().sort(function (a, b) { return b.puntaje - a.puntaje; });
  const top = secs[0], low = secs[secs.length - 1];
  const detalleSec = function (s) { return dos ? " (mandos medios " + s.por.mm + ", Dirección " + s.por.cl + ")" : ""; };

  // 1. Portada
  let sl = nueva_(ctx, true);
  if (ctx.imgs.banner) sl.insertImage(ctx.imgs.banner, 0, 0, W, W * 239 / 1031);
  caja_(sl, (d.tipo === "cultura" ? "Colaboración y cultura en " : "Colaboración interfuncional en ") + d.clienteNombre, 36, 192, 648, 44, { size: 30, bold: true, color: COL.white });
  caja_(sl, "Resultados de la encuesta · " + d.fecha, 36, 240, 648, 22, { size: 14, color: COL.sky });
  caja_(sl, gs.map(function (x) { return x.nombre + " (" + d.n[x.k].n + ")"; }).join(" · "), 36, 262, 648, 22, { size: 14, color: COL.sand });

  // 2. Tres mensajes
  sl = contenido_(ctx, "Tres mensajes principales");
  let m1;
  if (dos) {
    const mmMas = d.secciones.filter(function (s) { return s.por.mm - s.por.cl >= 5; }).length;
    const clMas = d.secciones.filter(function (s) { return s.por.cl - s.por.mm >= 5; }).length;
    const N = d.secciones.length;
    if (!mmMas && !clMas) m1 = ["Una mirada compartida", "Mandos medios y Dirección dan puntajes parecidos en todas las dimensiones."];
    else if (mmMas >= clMas) m1 = ["Dos miradas distintas", "Los mandos medios ven la colaboración de forma más positiva que Dirección en " + mmMas + " de las " + N + " dimensiones."];

    });

function pie(counts, colors){
  var tot = counts.reduce(function(a,b){return a+b;},0), a = -Math.PI/2, out = "", r = 52, cx = 60, cy = 60;
  if (!tot) return '<svg viewBox="0 0 120 120"><circle cx="60" cy="60" r="52" fill="var(--line)"/><text x="60" y="65" text-anchor="middle" font-size="12" fill="var(--muted)">Sin datos</text></svg>';
  counts.forEach(function(c,j){ if(!c) return; var f = c/tot;
    if (f >= 0.9999) { out += '<circle cx="60" cy="60" r="52" fill="'+colors[j]+'"/>'; return; }
    var a2 = a + f*2*Math.PI, x1 = cx+r*Math.cos(a), y1 = cy+r*Math.sin(a), x2 = cx+r*Math.cos(a2), y2 = cy+r*Math.sin(a2);
    out += '<path d="M60 60 L'+x1.toFixed(2)+' '+y1.toFixed(2)+' A52 52 0 '+(f>0.5?1:0)+' 1 '+x2.toFixed(2)+' '+y2.toFixed(2)+' Z" fill="'+colors[j]+'" stroke="#fff" stroke-width="2"><title>'+Math.round(f*100)+'%</title></path>';
    var mid = (a+a2)/2; if (f >= 0.08) out += '<text x="'+(cx+r*0.62*Math.cos(mid)).toFixed(1)+'" y="'+(cy+r*0.62*Math.sin(mid)+4).toFixed(1)+'" text-anchor="middle" font-size="11" font-weight="600" fill="'+(j===1||j===2&&counts.length===4?"#15201e":"#fff")+'">'+Math.round(f*100)+'%</text>';
    a = a2; });
  return '<svg viewBox="0 0 120 120" role="img">'+out+'</svg>';
}
function scores(s){ var by = {}, order = [];
  s.preguntas.forEach(function(p){ if(!p.opciones) return; var n = p.conteos.reduce(function(a,b){return a+b;},0); if(!n) return;
    var sc = p.conteos.reduce(function(a,c,j){ return a + c*pts(j,p.opciones.length); },0)/n;
    if(!by[p.seccion]) { by[p.seccion] = []; order.push(p.seccion); } by[p.seccion].push(sc); });
  return order.map(function(k){ return [k, Math.round(by[k].reduce(function(a,b){return a+b;},0)/by[k].length)]; });
}

      html += '<div class="pies">' + ok.map(function(s){ var q = s.preguntas[i]; var n = q.conteos.reduce(function(a,b){return a+b;},0);

    html += '</div>';
  });

}
