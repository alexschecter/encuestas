  const config = { catalogo: CATALOGO, asunto: fijo(props.getProperty("asunto")) || ASUNTO_INICIAL, mensaje: fijo(props.getProperty("mensaje")) || MENSAJE_INICIAL,

    asuntoInicial: ASUNTO_INICIAL, mensajeInicial: MENSAJE_INICIAL, firma: leerFirma_() };
  return HtmlService.createHtmlOutput(PAGINA.replace("__CONFIG__", JSON.stringify(config).replace(/</g, "\\u003c")))
    .setTitle("Dashboard")
    .addMetaTag("viewport", "width=device-width, initial-scale=1");

function buscarFormulario_(enc) {
  const props = PropertiesService.getScriptProperties();
  const guardado = props.getProperty(enc.id);
  if (guardado) { try { return FormApp.openById(guardado); } catch (e) {} }
  const archivos = DriveApp.getFilesByType(MimeType.GOOGLE_FORMS);
  while (archivos.hasNext()) {
    const f = archivos.next();
    if (f.isTrashed()) continue;
    try {
      const form = FormApp.openById(f.getId());
      if (form.getPublishedUrl().indexOf(enc.publicada) >= 0) { props.setProperty(enc.id, f.getId()); return form; }
    } catch (e) {}
  }
  return null;
}

// Datos de tu firma (web, email, teléfono, LinkedIn). Se escriben en el dashboard y se guardan solo en tu cuenta.
function leerFirma_() {
  try { return JSON.parse(PropertiesService.getUserProperties().getProperty("firma") || "{}"); } catch (e) { return {}; }
}
function guardarFirma(json) {
  const f = JSON.parse(json || "{}"), limpio = {};
  ["web", "email", "tel", "linkedin"].forEach(function (k) { limpio[k] = String(f[k] || "").trim(); });
  PropertiesService.getUserProperties().setProperty("firma", JSON.stringify(limpio));
  return true;
}
function escHtml_(t) { return String(t).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;"); }

// Arma la versión con formato del email: el texto, el enlace como botón y tu firma con el logo.
function emailHtml_(cuerpo, f) {
  const azul = "#4A90D9", gris = "#555555";
  const parrafos = String(cuerpo).split(/\n\s*\n/).map(function (p) {
    const t = p.trim();
    if (/^https?:\/\/\S+$/.test(t)) return '<p style="margin:22px 0"><a href="' + escHtml_(t) + '" style="background:' + azul + ';color:#ffffff;text-decoration:none;padding:12px 22px;border-radius:8px;font-weight:bold;display:inline-block">Responder la encuesta</a></p>' +
      '<p style="margin:0 0 16px;font-size:12px;color:#888888">Si el botón no funciona, copiá este enlace: <a href="' + escHtml_(t) + '" style="color:' + azul + '">' + escHtml_(t) + '</a></p>';
    return '<p style="margin:0 0 14px">' + escHtml_(t).replace(/(https?:\/\/[^\s<]+)/g, '<a href="$1" style="color:' + azul + '">$1</a>').replace(/\n/g, "<br>") + '</p>';
  }).join("");
  const fila = function (icono, txt, href) { return txt ? '<tr><td style="padding:1px 6px 1px 0;color:' + azul + ';font-size:12px">' + icono + '</td><td style="padding:1px 0;font-size:12px"><a href="' + escHtml_(href) + '" style="color:' + gris + ';text-decoration:none">' + escHtml_(txt) + '</a></td></tr>' : ""; };
  const web = f.web ? f.web.replace(/^https?:\/\//, "") : "", li = f.linkedin ? f.linkedin.replace(/^https?:\/\/(www\.)?linkedin\.com\/in\//, "").replace(/\/$/, "") : "";
  const datos = fila("&#127760;", web, "https://" + web) + fila("&#9993;", f.email, "mailto:" + f.email) + fila("&#9742;", f.tel, "tel:" + String(f.tel || "").replace(/[^+\d]/g, "")) + fila("in", li, "https://www.linkedin.com/in/" + li);
  const firma = '<table cellpadding="0" cellspacing="0" border="0" style="margin-top:26px"><tr>' +
    '<td style="padding-right:16px;vertical-align:middle"><img src="cid:logo" width="144" height="57" alt="Alex Schecter Coaching" style="display:block;border:0;width:144px;height:57px;max-width:144px"></td>' +
    (datos ? '<td style="vertical-align:middle;border-left:1px solid #e3e3e3;padding-left:14px"><table cellpadding="0" cellspacing="0" border="0">' + datos + '</table></td>' : "") + '</tr></table>';
  return '<div style="font-family:Helvetica,Arial,sans-serif;font-size:14px;line-height:1.5;color:#222222;max-width:560px">' + parrafos + firma + '</div>';
}
function firmaTexto_(f) {
  const lineas = [f.web, f.email, f.tel, f.linkedin ? "LinkedIn: " + f.linkedin : ""].filter(Boolean);
  return "\n\n--\nAlex Schecter Coaching" + (lineas.length ? "\n" + lineas.join("\n") : "");
}

// Envía el mensaje desde tu cuenta de Gmail, un email por persona.
// "desde" puede ser tu Gmail o cualquier dirección que tengas configurada en Gmail como "Enviar como" (por ejemplo, la de tu dominio).
function enviarEmails(destinatarios, asunto, cuerpo, desde, nombre, responderA, firmaJson) {
  if (firmaJson) guardarFirma(firmaJson);
  const lista = String(destinatarios || "").split(/[\s,;]+/).map(function (x) { return x.trim(); }).filter(Boolean);
  const validos = lista.filter(function (x) { return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(x); });
  const invalidos = lista.filter(function (x) { return validos.indexOf(x) < 0; });
  if (!validos.length) throw new Error("No encontré ninguna dirección de email válida.");
  const cupo = MailApp.getRemainingDailyQuota();
  if (validos.length > cupo) throw new Error("Hoy podés enviar " + cupo + " emails más desde el tablero (límite diario de Google).");
  const opciones = {};
  if (desde && GmailApp.getAliases().indexOf(desde) >= 0) opciones.from = desde;
  if (nombre) opciones.name = String(nombre);
  if (responderA && /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(responderA)) opciones.replyTo = String(responderA).trim();
  PropertiesService.getUserProperties().setProperty("desde", desde || "");
  const f = leerFirma_();
  opciones.htmlBody = emailHtml_(cuerpo, f);
  try { opciones.inlineImages = { logo: UrlFetchApp.fetch(IMG_BASE + "firma-logo.png").getBlob().setName("logo.png") }; }
  catch (e) { opciones.htmlBody = opciones.htmlBody.replace(/<img [^>]*>/, '<b style="font-size:16px;color:#111A27">Alex Schecter</b><br><span style="font-size:11px;letter-spacing:2px;color:#555555">COACHING</span>'); }
  const texto = cuerpo + firmaTexto_(f);
  validos.forEach(function (to) { GmailApp.sendEmail(to, asunto, texto, opciones); });
  return JSON.stringify({ enviados: validos.length, invalidos: invalidos });
}

const DEFS_URL = "https://alexschecter.github.io/encuestas/encuestas.json";
function definiciones_() {
  const cache = CacheService.getScriptCache(), hit = cache.get("defs");
  if (hit) return JSON.parse(hit);
  const t = UrlFetchApp.fetch(DEFS_URL + "?t=" + Date.now()).getContentText();
  try { cache.put("defs", t, 600); } catch (e) {}
  return JSON.parse(t);
}

// Ejecutar UNA vez desde el editor: crea un Google Form (con su hoja de respuestas) por cada encuesta nueva
// y muestra en el registro los datos que la página de la encuesta necesita para enviar las respuestas.
// Si se vuelve a ejecutar, no duplica nada: reutiliza los formularios ya creados.
function crearEncuestasNuevas() {
  const props = PropertiesService.getScriptProperties();
  definiciones_().forEach(function (e) {
    let form = null;
    const guardado = props.getProperty("nueva:" + e.id);
    if (guardado) { try { form = FormApp.openById(guardado); } catch (x) {} }
    if (!form) {
      form = FormApp.create(e.titulo);
      form.setDescription(e.intro);
      form.setCollectEmail(false);
      e.secciones.forEach(function (sec) {
        sec.preguntas.forEach(function (q) {
          let it;
          if (q.tipo === "texto") it = form.addParagraphTextItem();
          else if (q.tipo === "elegir3") it = form.addCheckboxItem().setChoiceValues(q.opciones);
          else it = form.addMultipleChoiceItem().setChoiceValues(q.opciones);
          it.setTitle(q.titulo).setHelpText(q.id);
        });
      });
      if (e.rolDesdeEnlace) form.addMultipleChoiceItem().setTitle("Rol").setHelpText("ROL").setChoiceValues(["Dirección", "Mandos medios"]);
      form.addTextItem().setTitle("Nombre (opcional)").setHelpText("NOMBRE");
      form.addTextItem().setTitle(CAMPO_CLIENTE).setHelpText("CLIENTE");
      const hoja = SpreadsheetApp.create(e.titulo + " (respuestas)");
      form.setDestination(FormApp.DestinationType.SPREADSHEET, hoja.getId());
      if (form.setPublished) form.setPublished(true);
      props.setProperty("nueva:" + e.id, form.getId());
    }
    const entries = {};
    form.getItems().forEach(function (it) {
      const t = it.getType(); let r = null;
      if (t === FormApp.ItemType.MULTIPLE_CHOICE) { const m = it.asMultipleChoiceItem(); r = m.createResponse(m.getChoices()[0].getValue()); }
      else if (t === FormApp.ItemType.CHECKBOX) { const c = it.asCheckboxItem(); r = c.createResponse([c.getChoices()[0].getValue()]); }
      else if (t === FormApp.ItemType.PARAGRAPH_TEXT) r = it.asParagraphTextItem().createResponse("x");
      else if (t === FormApp.ItemType.TEXT) r = it.asTextItem().createResponse("x");
      if (!r) return;
      const m = form.createResponse().withItemResponse(r).toPrefilledUrl().match(/entry\.(\d+)=/);
      if (m) entries[it.getHelpText()] = +m[1];
    });
    const pub = form.getPublishedUrl().match(/\/d\/e\/([^/]+)/);
    Logger.log("ENCUESTA " + JSON.stringify({ id: e.id, formId: pub ? pub[1] : form.getId(), entries: entries }));
  });
  Logger.log("Listo. Copiá todo este registro y pegalo en el chat.");
}

// Lee las respuestas de las encuestas nuevas, separadas por grupo (mandos medios / Dirección) y cliente.
function datosNuevas_() {
  const props = PropertiesService.getScriptProperties(), res = [];
  let defs = [];
  try { defs = definiciones_(); } catch (e) { return res; }
  defs.forEach(function (e) {
    const fid = props.getProperty("nueva:" + e.id);
    if (!fid) return;
    let form;
    try { form = FormApp.openById(fid); } catch (x) { return; }
    const preguntas = [], secciones = [];
    e.secciones.forEach(function (sec) {
      secciones.push({ titulo: sec.titulo, desc: sec.desc || null, area: !!sec.area, perfil: !!sec.perfil });
      sec.preguntas.forEach(function (q) {
        preguntas.push({ id: q.id, seccion: sec.titulo, titulo: q.titulo, corto: q.corto || null, tipo: q.tipo,
          opciones: q.tipo === "texto" ? null : q.opciones, puntajes: q.puntajes || null, multi: q.tipo === "elegir3", perfil: !!sec.perfil });
      });
    });
    const pos = {};
    preguntas.forEach(function (p, i) { pos[p.id] = i; });
    const porGrupo = { mm: {}, cl: {} };
    form.getResponses().forEach(function (r) {
      let cliente = "", rol = e.grupo || "mm", nombre = "";
      const vals = {};
      r.getItemResponses().forEach(function (ir) {
        const h = ir.getItem().getHelpText(), v = ir.getResponse();
        if (h === "CLIENTE") cliente = String(v || "").trim();
        else if (h === "ROL") rol = v === "Dirección" ? "cl" : "mm";
        else if (h === "NOMBRE") nombre = String(v || "").trim();
        else vals[h] = v;
      });
      cliente = cliente || "(sin cliente)";
      const clientes = porGrupo[rol];
      const c = clientes[cliente] || (clientes[cliente] = { n: 0, ultima: null, conteos: preguntas.map(function (p) { return p.opciones ? p.opciones.map(function () { return 0; }) : null; }), textos: preguntas.map(function (p) { return p.opciones ? null : []; }) });
      c.n++;
      if (e.tipo === "desarrollo") (c.personas || (c.personas = [])).push({ nombre: nombre, r: vals });
      const ts = r.getTimestamp(); if (!c.ultima || ts > c.ultima) c.ultima = ts;
      Object.keys(vals).forEach(function (h) {
        const i = pos[h]; if (i === undefined) return;
        const p = preguntas[i], v = vals[h];
        if (p.multi) (Array.isArray(v) ? v : [v]).forEach(function (x) { const j = p.opciones.indexOf(x); if (j >= 0) c.conteos[i][j]++; });
        else if (p.opciones) { const j = p.opciones.indexOf(v); if (j >= 0) c.conteos[i][j]++; }
        else if (v && String(v).trim()) c.textos[i].push(String(v).trim());
      });
    });
    ["mm", "cl"].forEach(function (g) {
      if (e.grupo && e.grupo !== g) return;
      const clientes = porGrupo[g];
      Object.keys(clientes).forEach(function (k) { clientes[k].ultima = clientes[k].ultima ? clientes[k].ultima.toISOString() : null; });
      res.push({ id: e.id + (g === "cl" && !e.grupo ? "-direccion" : ""), grupo: g, tipo: "n-" + e.id, titulo: e.nombre, nueva: true, clase: e.tipo, tema: e.tema || null,
        secciones: secciones, preguntas: preguntas, clientes: clientes });
    });
  });
  return res;
}

function datosColaboracion_() {
  return ENCUESTAS.map(function (enc) {
    const form = buscarFormulario_(enc);
    if (!form) return { id: enc.id, grupo: enc.grupo, tipo: enc.tipo, titulo: enc.titulo, error: "No encontré este formulario en tu Drive." };
    const respuestas = form.getResponses();
    let seccion = "", idCliente = null;
    const preguntas = [];
    form.getItems().forEach(function (item) {
      const t = item.getType();
      if (t === FormApp.ItemType.PAGE_BREAK) { seccion = item.getTitle(); return; }
      if (item.getTitle() === CAMPO_CLIENTE) { idCliente = item.getId(); return; }
      if (item.getTitle() === "Nombre (opcional)") return; // los nombres no se muestran en el tablero
      if (t === FormApp.ItemType.MULTIPLE_CHOICE) {
        const opciones = item.asMultipleChoiceItem().getChoices().map(function (c) { return c.getValue(); });
        preguntas.push({ id: item.getId(), seccion: seccion, titulo: item.getTitle(), opciones: opciones });
      } else if (t === FormApp.ItemType.PARAGRAPH_TEXT || t === FormApp.ItemType.TEXT) {
        preguntas.push({ id: item.getId(), seccion: seccion, titulo: item.getTitle(), opciones: null });
      }
    });
    const pos = {};
    preguntas.forEach(function (p, i) { pos[p.id] = i; });
    const clientes = {};
    respuestas.forEach(function (r) {
      let cliente = "";
      const its = r.getItemResponses();
      its.forEach(function (ir) { if (ir.getItem().getId() === idCliente) cliente = String(ir.getResponse() || "").trim(); });
      cliente = cliente || "(sin cliente)";
      const c = clientes[cliente] || (clientes[cliente] = { n: 0, ultima: null, conteos: preguntas.map(function (p) { return p.opciones ? p.opciones.map(function () { return 0; }) : null; }), textos: preguntas.map(function (p) { return p.opciones ? null : []; }) });
      c.n++;
      const ts = r.getTimestamp(); if (!c.ultima || ts > c.ultima) c.ultima = ts;
      its.forEach(function (ir) {
        const i = pos[ir.getItem().getId()]; if (i === undefined) return;
        const p = preguntas[i], v = ir.getResponse();
        if (p.opciones) { const j = p.opciones.indexOf(v); if (j >= 0) c.conteos[i][j]++; }
        else if (v && String(v).trim()) c.textos[i].push(String(v).trim());
      });
    });
    Object.keys(clientes).forEach(function (k) { clientes[k].ultima = clientes[k].ultima ? clientes[k].ultima.toISOString() : null; });
    return { id: enc.id, grupo: enc.grupo, tipo: enc.tipo, titulo: enc.tipo === "cultura" ? "Colaboración & Cultura" : "Colaboración", preguntas: preguntas, clientes: clientes };
  });
}

  Logger.log("Podés enviar desde: " + [Session.getEffectiveUser().getEmail()].concat(GmailApp.getAliases()).join(", "));

function generarInforme(cliente, tipo) {
  if (!cliente || cliente === "(sin cliente)") throw new Error("Elegí un cliente en el menú.");
  const raw = JSON.parse(obtenerDatos());
  const d = prepararDatos_(raw, cliente, tipo || "colab");
  const nombreBase = d.clienteNombre + " · " + d.tituloEncuesta + " · " + d.fecha;
  const carpeta = carpetaInformes_();

  const pres = SlidesApp.create(nombreBase + " · Completo");
  const ctx = { pres: pres, d: d, pag: 0, imgs: imagenes_() };
  const inicial = pres.getSlides()[0];
  if (d.clase === "desarrollo") { construirDesarrollo_(ctx); construirDetalleDesarrollo_(ctx); }
  else { construirResumen_(ctx); construirDetalle_(ctx); }
  const nResumen = ctx.nCerradas; // el Resumen incluye las tortas por pregunta, sin las respuestas abiertas
  inicial.remove();
  pres.saveAndClose();
  const fCompleto = DriveApp.getFileById(pres.getId());
  fCompleto.moveTo(carpeta);

  const fResumen = fCompleto.makeCopy(nombreBase + " · Resumen", carpeta);
  const pr = SlidesApp.openById(fResumen.getId());
  pr.getSlides().slice(nResumen).forEach(function (s) { s.remove(); });
  pr.saveAndClose();

  const pdf = function (f) { return carpeta.createFile(f.getAs(MimeType.PDF)).setName(f.getName() + ".pdf").getUrl(); };
  return JSON.stringify({
    resumen: { url: fResumen.getUrl(), pdf: pdf(fResumen) },
    completo: { url: fCompleto.getUrl(), pdf: pdf(fCompleto) },
  });
}

function carpetaInformes_() {
  const it = DriveApp.getFoldersByName(CARPETA_INFORMES);
  return it.hasNext() ? it.next() : DriveApp.createFolder(CARPETA_INFORMES);
}

function imagenes_() {
  const get = function (n) { try { return UrlFetchApp.fetch(IMG_BASE + n).getBlob(); } catch (e) { return null; } };
  return { banner: get("banner.jpg"), logo: get("logo.png"), firma: get("firma-logo.png") };
}

// Gráfico de torta (imagen) con el % de cada respuesta de un grupo. Si Google no puede dibujarlo, devuelve null.
function torta_(q, k, cols) {
  const c = q.por[k].conteos;
  if (!c.reduce(function (a, b) { return a + b; }, 0)) return null;
  try {
    const dt = Charts.newDataTable().addColumn(Charts.ColumnType.STRING, "Respuesta").addColumn(Charts.ColumnType.NUMBER, "Personas");
    q.opciones.forEach(function (o, i) { dt.addRow([o, c[i]]); });
    return Charts.newPieChart().setDataTable(dt.build()).setColors(cols).setDimensions(360, 360)
      .setLegendPosition(Charts.Position.NONE).setOption("pieSliceText", "percentage")
      .setOption("pieSliceTextStyle", { fontSize: 22, color: "#FFFFFF" }).setOption("pieSliceBorderColor", "#FFFFFF")
      .setOption("chartArea", { left: 6, top: 6, width: "96%", height: "96%" }).setOption("backgroundColor", "#FFFFFF")
      .build().getAs("image/png");
  } catch (e) { return null; }
}

// Una pregunta con una torta por grupo y el % de cada respuesta (alto: 150).
function bloquePregunta_(ctx, sl, q, y) {
  const gs = ctx.d.grupos, cols = ESCALA[q.opciones.length] || ESCALA[4];
  const tl = q.titulo.length > 95;
  caja_(sl, q.titulo, 30, tl ? y - 6 : y, 660, tl ? 40 : 34, { size: tl ? 11.5 : 13, bold: true, color: COL.navy });
  gs.forEach(function (g, gi) {
    const x0 = 30 + gi * 335, c = q.por[g.k].conteos, n = c.reduce(function (a, b) { return a + b; }, 0);
    caja_(sl, g.nombre, x0, y + 34, 320, 20, { size: 11, bold: true, color: g.color });
    const img = torta_(q, g.k, cols);
    if (img) sl.insertImage(img, x0, y + 56, 90, 90);
    else apilada_(sl, q, g.k, x0, y + 90, 90, 20);
    const largo = q.opciones.some(function (o) { return o.length > 30; });
    const lineas = q.opciones.map(function (o, k) { return "●  " + pct_(c[k], n) + "%  " + o; });
    const ley = caja_(sl, lineas.join("\n"), x0 + 96, y + 54, 236, 72, { size: largo ? 8.5 : 10 }).getText();
    let p = 0;
    lineas.forEach(function (t, k) { ley.getRange(p, p + 1).getTextStyle().setForegroundColor(cols[k] || COL.mute); p += t.length + 1; });
    const v = q.por[g.k].puntaje;
    caja_(sl, "Puntaje " + (v === null ? "–" : v + "/100 · " + banda_(v).t), x0 + 96, y + 126, 230, 20, { size: 10, color: COL.mute });
  });
}
// Varias preguntas, de a 2 por diapositiva, con el mismo título.
function paginasPreguntas_(ctx, titulo, sub, lista) {
  for (let i = 0; i < lista.length; i += 2) {
    const sl = contenido_(ctx, titulo, sub);
    lista.slice(i, i + 2).forEach(function (q, j) { bloquePregunta_(ctx, sl, q, (sub ? 80 : 66) + j * 148); });
  }
}

// ── Datos ──
function puntos_(j, n) { return n > 1 ? (n - 1 - j) / (n - 1) * 100 : 100; }
function puntaje_(conteos, puntajes) {
  const n = conteos.reduce(function (a, b) { return a + b; }, 0);
  if (!n) return null;
  return Math.round(conteos.reduce(function (a, c, j) { return a + c * (puntajes ? puntajes[j] : puntos_(j, conteos.length)); }, 0) / n);
}
const TODOS = { k: "todos", nombre: "Todos", color: COL.mute };
function promedio_(xs) { xs = xs.filter(function (x) { return x !== null && x !== undefined; }); return xs.length ? Math.round(xs.reduce(function (a, b) { return a + b; }, 0) / xs.length) : null; }
// Escala única para todo el informe: [rango, estadio, mínimo, color]
const ESTADIOS = [["76-100", "Conectado y estratégico", 76, COL.blue], ["51-75", "Funcional pero fragmentado", 51, COL.sky], ["26-50", "Transaccional y a veces tenso", 26, COL.sand], ["0-25", "Aislado e ineficiente", 0, COL.coral]];
function estadio_(v) { return ESTADIOS.filter(function (e) { return v >= e[2]; })[0]; }
// Para preguntas y dimensiones se usan nombres simples con los mismos rangos y colores.
const NIVELES = ["Fortaleza", "Funcional", "A mejorar", "Crítico"];
function banda_(s) { if (s === null || s === undefined) return { t: "Sin datos", c: COL.line }; const e = estadio_(s); return { t: NIVELES[ESTADIOS.indexOf(e)], c: e[3] }; }

function prepararDatos_(raw, cliente, tipo) {
  const sets = raw.filter(function (s) { return s.tipo === tipo && !s.error; });
  if (!sets.length) throw new Error("No encontré esta encuesta.");
  const meta = sets[0], nueva = !!meta.nueva, clase = meta.clase || "colab";
  const g = {};
  sets.forEach(function (s) {
    const c = s.clientes[cliente];
    g[s.grupo] = { n: c ? c.n : 0, personas: c && c.personas || [], preguntas: s.preguntas.map(function (p, i) {
      return { seccion: p.seccion, titulo: p.titulo, opciones: p.opciones, conteos: c ? c.conteos[i] : (p.opciones ? p.opciones.map(function () { return 0; }) : null), textos: c ? c.textos[i] : [] };
    }) };
  });
  const conDatos = GRUPOS.filter(function (x) { return g[x.k] && g[x.k].n > 0; });
  if (!conDatos.length) throw new Error("Todavía no hay respuestas de este cliente en esta encuesta.");
  // Encuestas nuevas por dimensiones: un grupo se muestra solo con 3 respuestas o más; si no, se suma a "Todos".
  let grupos = conDatos, insuf = [];
  if (nueva && clase === "dimensiones") {
    const ok = conDatos.filter(function (x) { return g[x.k].n >= 3; });
    insuf = conDatos.filter(function (x) { return g[x.k].n < 3; });
    if (ok.length < 2) {
      const todos = { n: 0, preguntas: null };
      conDatos.forEach(function (x) { todos.n += g[x.k].n; });
      todos.preguntas = g[conDatos[0].k].preguntas.map(function (p, i) {
        const conteos = p.opciones ? p.opciones.map(function (o, j) { return conDatos.reduce(function (a, x) { return a + g[x.k].preguntas[i].conteos[j]; }, 0); }) : null;
        // En "Todos" van solo las respuestas abiertas de los grupos que no se muestran aparte (para no repetirlas).
        const textos = [].concat.apply([], insuf.map(function (x) { return g[x.k].preguntas[i].textos || []; }));
        return { conteos: conteos, textos: textos };
      });
      g.todos = todos;
      grupos = ok.concat([TODOS]);
    }
  }
  const base = g[conDatos[0].k].preguntas;
  const defP = meta.preguntas || [];
  const preguntas = base.map(function (p, i) {
    const dp = defP[i] || {};
    const q = { seccion: p.seccion, titulo: p.titulo, opciones: p.opciones, puntajes: dp.puntajes || null, multi: !!dp.multi, perfil: !!dp.perfil, corto: dp.corto || null, id: dp.id || null, por: {} };
    const conPuntaje = p.opciones && !q.multi && !q.perfil;
    grupos.forEach(function (x) { const pq = g[x.k].preguntas[i]; q.por[x.k] = { conteos: pq.conteos, textos: pq.textos || [], puntaje: conPuntaje ? puntaje_(pq.conteos, q.puntajes) : null }; });
    if (conPuntaje) q.puntaje = promedio_(grupos.map(function (x) { return q.por[x.k].puntaje; }));
    return q;
  });
  const secciones = [];
  preguntas.forEach(function (q) {
    if (!q.opciones || q.multi || q.perfil) return;
    let s = secciones.filter(function (x) { return x.titulo === q.seccion; })[0];
    if (!s) { s = { titulo: q.seccion, qs: [] }; secciones.push(s); }
    s.qs.push(q);
  });
  secciones.forEach(function (s) {
    s.por = {};
    grupos.forEach(function (x) { s.por[x.k] = promedio_(s.qs.map(function (q) { return q.por[x.k].puntaje; })); });
    s.puntaje = promedio_(grupos.map(function (x) { return s.por[x.k]; }));
  });
  const meses = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
  const hoy = new Date();
  const nombre = String(cliente).replace(/[-_]+/g, " ").replace(/\b\w/g, function (c) { return c.toUpperCase(); });
  const descs = {};
  (meta.secciones || []).forEach(function (x) { if (x.desc) descs[x.titulo] = x.desc; });
  return {
    cliente: cliente, clienteNombre: nombre, tipo: tipo, grupos: grupos, n: g, preguntas: preguntas, secciones: secciones,
    nueva: nueva, clase: clase, insuf: insuf, descs: descs,
    tema: meta.tema || "la colaboración",
    dos: grupos.length === 2 && !!g.mm && !!g.cl && grupos.every(function (x) { return x.k !== "todos"; }),
    tituloEncuesta: nueva ? meta.titulo : tipo === "cultura" ? "Colaboración Interfuncional y Cultura" : "Colaboración Interfuncional",
    fecha: meses[hoy.getMonth()] + " " + hoy.getFullYear(),
  };
}

// ── Dibujo ──
function caja_(sl, text, l, t, w, h, o) {
  o = o || {};
  const s = sl.insertTextBox(text || " ", l, t, w, h);
  const tr = s.getText();
  tr.getTextStyle().setFontFamily(FONT).setFontSize(o.size || 14).setForegroundColor(o.color || COL.ink).setBold(!!o.bold).setItalic(!!o.italic);
  if (o.align) tr.getParagraphStyle().setParagraphAlignment(o.align === "center" ? SlidesApp.ParagraphAlignment.CENTER : o.align === "end" ? SlidesApp.ParagraphAlignment.END : SlidesApp.ParagraphAlignment.START);
  s.setContentAlignment(o.v === "middle" ? SlidesApp.ContentAlignment.MIDDLE : o.v === "bottom" ? SlidesApp.ContentAlignment.BOTTOM : SlidesApp.ContentAlignment.TOP);
  return s;
}
function forma_(sl, tipo, l, t, w, h, color) {
  const s = sl.insertShape(tipo, l, t, w, h);
  s.getFill().setSolidFill(color);
  s.getBorder().setTransparent();
  return s;
}
function rect_(sl, l, t, w, h, color) { return forma_(sl, SlidesApp.ShapeType.RECTANGLE, l, t, w, h, color); }
function tarjeta_(sl, l, t, w, h) { return forma_(sl, SlidesApp.ShapeType.ROUND_RECTANGLE, l, t, w, h, COL.mist); }
function circulo_(sl, l, t, d, color, text, size) {
  const s = forma_(sl, SlidesApp.ShapeType.ELLIPSE, l, t, d, d, color);
  if (text) {
    const tr = s.getText(); tr.setText(text);
    tr.getTextStyle().setFontFamily(FONT).setFontSize(size || 14).setBold(true).setForegroundColor(COL.white);
    tr.getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER);
    s.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE);
  }
  return s;
}
function nueva_(ctx, oscura) {
  const sl = ctx.pres.appendSlide(SlidesApp.PredefinedLayout.BLANK);
  sl.getBackground().setSolidFill(oscura ? COL.navy : COL.white);
  ctx.pag++;
  return sl;
}
function contenido_(ctx, titulo, sub) {
  const sl = nueva_(ctx, false);
  caja_(sl, titulo, 30, 18, 570, 40, { size: 24, bold: true, color: COL.navy });
  if (sub) caja_(sl, sub, 30, 54, 570, 22, { size: 12, color: COL.mute });
  if (ctx.imgs.firma) sl.insertImage(ctx.imgs.firma, 614, 16, 76, 30);
  caja_(sl, "Encuesta de " + ctx.d.tituloEncuesta + " · " + ctx.d.clienteNombre, 30, 376, 520, 20, { size: 9, color: COL.mute });
  caja_(sl, String(ctx.pag), 640, 376, 50, 20, { size: 9, color: COL.mute, align: "end" });
  return sl;
}
function pct_(c, n) { return n ? Math.round(c / n * 100) : 0; }
function nombreGrupo_(k) { return GRUPOS.concat([TODOS]).filter(function (x) { return x.k === k; })[0].nombre; }
function masElegida_(q, k) {
  const c = q.por[k].conteos; let j = 0;
  c.forEach(function (v, i) { if (v > c[j]) j = i; });
  return q.opciones[j];
}

// Barras horizontales por grupo (una fila por etiqueta)
function barras_(sl, d, filas, l, t, w, h) {
  // Cada fila: el nombre de la dimensión (y qué mide) arriba, y debajo una barra por grupo con su nombre a la izquierda.
  const gs = d.grupos, anchoG = 84;
  const alto = h / filas.length, barH = 11, maxW = w - anchoG - 96;
  filas.forEach(function (f, i) {
    const y = t + i * alto;
    const tit = caja_(sl, f.etiqueta + (f.desc ? "  ·  " + f.desc : ""), l - 4, y, w, 18, { size: 11, bold: true, color: COL.navy, v: "middle" }).getText();
    if (f.desc) tit.getRange(f.etiqueta.length, f.etiqueta.length + 5 + f.desc.length).getTextStyle().setBold(false).setFontSize(9.5).setForegroundColor(COL.mute);
    if (f.brecha) {
      const b = forma_(sl, SlidesApp.ShapeType.ROUND_RECTANGLE, l + w - 74, y + 19, 74, 2 * barH + 3, COL.coral);
      const bt = b.getText(); bt.setText("Brecha " + f.brecha);
      bt.getTextStyle().setFontFamily(FONT).setFontSize(9).setBold(true).setForegroundColor(COL.white);
      bt.getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER); b.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE);
    }
    gs.forEach(function (x, gi) {
      const v = f.valores[x.k], yy = y + 19 + gi * (barH + 3);
      caja_(sl, x.nombre, l - 4, yy - 4, anchoG, barH + 8, { size: 9, color: COL.mute, v: "middle" });
      if (v !== null && v > 0) rect_(sl, l + anchoG, yy, Math.max(2, maxW * v / 100), barH, x.color);
      const et = caja_(sl, v === null ? "–" : v + "  " + banda_(v).t, l + anchoG + Math.max(2, maxW * (v || 0) / 100) + 2, yy - 4, 96, barH + 8, { size: 9.5, color: COL.ink, v: "middle" }).getText();
      if (v !== null) et.getRange(0, String(v).length).getTextStyle().setBold(true);
    });
  });
}

// Barra apilada con el % de cada respuesta
function apilada_(sl, q, k, l, t, w, h) {
  const c = q.por[k].conteos, n = c.reduce(function (a, b) { return a + b; }, 0), cols = ESCALA[c.length] || ESCALA[4];
  if (!n) { rect_(sl, l, t, w, h, COL.line); return; }
  let x = l;
  c.forEach(function (v, j) {
    if (!v) return;
    const ww = w * v / n, s = rect_(sl, x, t, ww, h, cols[j]);
    if (ww >= 26) {
      const tr = s.getText(); tr.setText(pct_(v, n) + "%");
      tr.getTextStyle().setFontFamily(FONT).setFontSize(10).setBold(true).setForegroundColor(j === 0 || j === c.length - 1 ? COL.white : COL.ink);
      tr.getParagraphStyle().setParagraphAlignment(SlidesApp.ParagraphAlignment.CENTER);
      s.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE);
    }
    x += ww;
  });
}

// ── Resumen ──
function construirResumen_(ctx) {
  const d = ctx.d, gs = d.grupos, dos = d.dos;
  const secs = d.secciones.slice().sort(function (a, b) { return b.puntaje - a.puntaje; });
  const top = secs[0], low = secs[secs.length - 1];
  const detalleSec = function (s) { return dos ? " (mandos medios " + s.por.mm + ", Dirección " + s.por.cl + ")" : ""; };

  // 1. Portada
  let sl = nueva_(ctx, true);
  if (ctx.imgs.banner) sl.insertImage(ctx.imgs.banner, 0, 0, W, W * 239 / 1031);
  caja_(sl, (d.nueva ? d.tituloEncuesta + " en " : d.tipo === "cultura" ? "Colaboración y cultura en " : "Colaboración interfuncional en ") + d.clienteNombre, 36, d.nueva && d.tituloEncuesta.length > 26 ? 150 : 192, 648, d.nueva && d.tituloEncuesta.length > 26 ? 86 : 44, { size: 30, bold: true, color: COL.white, v: "bottom" });
  caja_(sl, "Resultados de la encuesta · " + d.fecha, 36, 240, 648, 22, { size: 14, color: COL.sky });
  caja_(sl, GRUPOS.filter(function (x) { return d.n[x.k] && d.n[x.k].n > 0; }).map(function (x) { return x.nombre + " (" + d.n[x.k].n + ")"; }).join(" · "), 36, 262, 648, 22, { size: 14, color: COL.sand });

  // 2. Tres mensajes
  sl = contenido_(ctx, "Tres mensajes principales");
  let m1;
  if (dos) {
    const mmMas = d.secciones.filter(function (s) { return s.por.mm - s.por.cl >= 5; }).length;
    const clMas = d.secciones.filter(function (s) { return s.por.cl - s.por.mm >= 5; }).length;
    const N = d.secciones.length;
    if (!mmMas && !clMas) m1 = ["Una mirada compartida", "Mandos medios y Dirección dan puntajes parecidos en todas las dimensiones."];
    else if (mmMas >= clMas) m1 = ["Dos miradas distintas", "Los mandos medios ven " + d.tema + " de forma más positiva que Dirección en " + mmMas + " de las " + N + " dimensiones."];
    else m1 = ["Dos miradas distintas", "Dirección ve " + d.tema + " de forma más positiva que los mandos medios en " + clMas + " de las " + N + " dimensiones."];
  } else if (gs.length === 1 && gs[0].k !== "todos") {
    m1 = ["Participación", "Respondieron " + d.n[gs[0].k].n + " personas de " + gs[0].nombre.toLowerCase() + "."];
  } else {
    const tot = GRUPOS.reduce(function (a, x) { return a + (d.n[x.k] ? d.n[x.k].n : 0); }, 0);
    m1 = ["Participación", "Respondieron " + tot + " personas. " + d.insuf.map(function (x) { return x.nombre; }).join(" y ") + (d.insuf.length > 1 ? " tienen" : " tiene") + " menos de 3 respuestas, por eso se muestra" + (d.insuf.length > 1 ? "n" : "") + " dentro de Todos."];
  }
  const msgs = [m1,
    ["Fortaleza: " + top.titulo, "Es la dimensión mejor valorada, con un puntaje de " + top.puntaje + detalleSec(top) + "."],
    ["Prioridad: " + low.titulo, "Es la dimensión con el puntaje más bajo: " + low.puntaje + detalleSec(low) + "."]];
  msgs.forEach(function (m, i) {
    const x = 30 + i * 222;
    tarjeta_(sl, x, 80, 210, 280);
    circulo_(sl, x + 16, 96, 36, COL.navy, String(i + 1), 16);
    caja_(sl, m[0], x + 12, 142, 186, 56, { size: 15, bold: true, color: COL.navy });
    caja_(sl, m[1], x + 12, 200, 186, 150, { size: 13 });
  });

  // 3. Participación y escala
  sl = contenido_(ctx, "Quiénes respondieron y cómo leer los resultados");
  GRUPOS.filter(function (x) { return d.n[x.k] && d.n[x.k].n > 0; }).forEach(function (x, i) {
    caja_(sl, String(d.n[x.k].n), 30 + i * 150, 80, 140, 70, { size: 54, bold: true, color: x.color });
    caja_(sl, x.nombre, 30 + i * 150, 150, 140, 22, { size: 14 });
  });
  const avisoInsuf = d.insuf.length ? d.insuf.map(function (x) { return x.nombre; }).join(" y ") + ": muestra insuficiente (menos de 3 respuestas). Para cuidar el anonimato, " + (d.insuf.length > 1 ? "sus respuestas se ven" : "sus respuestas se ven") + " solo dentro de Todos.\n\n" : "";

function datosDesarrollo_(d) {
  const k = d.grupos[0].k, gr = d.n[k];
  const habs = d.preguntas.filter(function (q) { return q.corto && q.opciones && !q.multi && !q.perfil; });
  const elegir = d.preguntas.filter(function (q) { return q.multi; })[0];
  const perfiles = d.preguntas.filter(function (q) { return q.perfil; });
  const abiertas = d.preguntas.filter(function (q) { return !q.opciones; });
  habs.forEach(function (q, i) {
    const c = q.por[k].conteos, n = c.reduce(function (a, b) { return a + b; }, 0);
    q.cuesta = n ? Math.round((c[2] + c[3]) / n * 100) : 0;
    q.pedidos = elegir ? elegir.por[k].conteos[i] || 0 : 0;   // el tema i de la lista corresponde a la habilidad i
    q.v = q.por[k].puntaje;
  });

  const porPedidos = habs.slice().sort(function (a, b) { return b.pedidos - a.pedidos || (a.v - b.v); });
  const bajas = porPuntaje.slice(0, 5), pedidas = porPedidos.filter(function (q) { return q.pedidos > 0; }).slice(0, 5);
  const claras = bajas.filter(function (q) { return pedidas.indexOf(q) >= 0; });
  return { k: k, gr: gr, habs: habs, elegir: elegir, perfiles: perfiles, abiertas: abiertas, porPuntaje: porPuntaje, porPedidos: porPedidos, bajas: bajas, pedidas: pedidas, claras: claras };
